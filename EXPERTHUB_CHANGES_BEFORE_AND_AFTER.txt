Pehle Express me koi central error handler nahi tha. Jab koi galat JSON bhejta tha
to server pura file path (/ExpertHub/backend/...) aur stack trace screen par leak karta tha.

--- [PEHLE (BEFORE)] ---
// Koi central error handler nahi tha.
// SyntaxError hone par Express ka default HTML error page leak ho jata tha.
app.get('/api/health', (req, res) => {
  res.json({ status: 'ok', message: 'Backend is running!' });
});

const PORT = process.env.PORT || 5000;

--- [ABHI (NOW)] ---
app.get('/api/health', (req, res) => {
  res.json({ status: 'ok', message: 'Backend is running!' });
});

// Centralized Error Handling Middleware (Suppresses raw stack trace / path leakage in production)
app.use((err, req, res, next) => {
  if (res.headersSent) {
    return next(err);
  }

  // Handle JSON parsing / body-parser syntax errors cleanly
  if (err instanceof SyntaxError && err.status === 400 && 'body' in err) {
    return res.status(400).json({
      success: false,
      message: 'Invalid JSON request payload format.'
    });
  }

  // Handle CORS rejection
  if (err.message && err.message.includes('CORS')) {
    return res.status(403).json({
      success: false,
      message: 'Access denied: Origin not authorized by CORS policy.'
    });
  }

  // Generic sanitized error response
  console.error('[UnhandledError]', err.message);
  res.status(err.status || 500).json({
    success: false,
    message: err.message && process.env.NODE_ENV === 'development'
      ? err.message
      : 'An unexpected internal error occurred. Please try again later.'
  });
});

const PORT = process.env.PORT || 5000;

FAYDA (BENEFIT):
Ab server kabhi bhi internal directory path ya code stack trace screen par print nahi
karega. Clean aur safe JSON error aayega.

// Line 185
const forgotPassword = async (req, res) => {
  try {
    const { email } = req.body || {};
    if (!email || typeof email !== 'string') {
      return res.status(400).json({ message: 'Please provide an email address' });
    }

    const normalizedEmail = email.toLowerCase().trim();
    const user = await User.findOne({ email: { $regex: new RegExp(`^${normalizedEmail}$`, 'i') } });
    const expert = await Expert.findOne({ email: { $regex: new RegExp(`^${normalizedEmail}$`, 'i') } });

    if (!user && !expert) {
      // Prevent user enumeration: Return generic safe response
      return res.status(200).json({
        success: true,
        message: 'If an account is associated with this email, a password reset link has been dispatched.'
      });
    }
    ...
    // Securely determine frontend URL (Never trust arbitrary unvalidated client headers)
    const allowedFrontends = [
      'https://experthub.nextgoodtechnologies.in',
      'https://admin.experthub.nextgoodtechnologies.in',
      'http://localhost:5173',
      'http://localhost:5174',
      'http://localhost:3000'
    ];
    const incomingOrigin = req.get('origin') || req.get('referer');
    let validatedBase = process.env.FRONTEND_URL || 'https://experthub.nextgoodtechnologies.in';

    if (incomingOrigin) {
      try {
        const u = new URL(incomingOrigin);
        const originUrl = `${u.protocol}//${u.host}`;
        if (allowedFrontends.includes(originUrl)) {
          validatedBase = originUrl;
        }
      } catch (e) {}
    }

    const resetUrl = `${validatedBase}/reset-password/${resetToken}`;

1. Reset link hamesha verified official domain par hi banega, fake domain par nahi.
2. Anjaan emails par 404 ke bajaye safe generic response aayega taaki koi emails list verify na kar sake.
3. 'req.body || {}' lagane se empty POST par server crash hona band ho gaya.

router.get('/:id', async (req, res) => {
  try {
    const { id } = req.params;
    if (!id || id === 'public' || !mongoose.Types.ObjectId.isValid(id)) {
      return res.status(404).json({ message: 'Expert not found' });
    }

    const expert = await Expert.findById(id)
      .select(PUBLIC_EXPERT_FIELDS)
      .populate('categoryId', 'name')
      .populate('subCategoryId', 'name');
    if (expert) {
      res.json(expert);
    } else {
      res.status(404).json({ message: 'Expert not found' });
    }
  } catch (error) {
    res.status(500).json({ message: 'Server Error' });
  }
});
