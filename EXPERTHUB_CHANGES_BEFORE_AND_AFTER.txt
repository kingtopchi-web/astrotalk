1. FILE: backend/server.js (CORS Origin Restriction)
--------------------------------------------------------------------------------
PROBLEM:
Pehle server kisi bhi anjaan domain (jaise google.com ya malicious-site.com) ko
CORS allow kar deta tha aur unhe credentials ke sath data de deta tha.

--- [PEHLE (BEFORE)] ---
// CORS Configuration
const allowedOrigins = (process.env.ALLOWED_ORIGINS || 'https://experthub.nextgoodtechnologies.in,http://localhost:5173,http://localhost:5174,http://localhost:3000').split(',');

app.use(cors({
  origin: function (origin, callback) {
    if (!origin) return callback(null, true);
    if (allowedOrigins.indexOf(origin) !== -1 || process.env.NODE_ENV !== 'production') {
      return callback(null, true);
    }
    return callback(new Error('Not allowed by CORS'));
  },
  credentials: true
}));

--- [ABHI (NOW)] ---
// CORS Configuration - Strictly restricted to authorized platform domains (Zero localhost in default code)
const defaultAllowed = [
  'https://experthub.nextgoodtechnologies.in',
  'https://admin.experthub.nextgoodtechnologies.in'
];
const envAllowed = process.env.ALLOWED_ORIGINS ? process.env.ALLOWED_ORIGINS.split(',').map(s => s.trim()) : [];
const allowedOrigins = Array.from(new Set([...defaultAllowed, ...envAllowed])).filter(Boolean);

app.use(cors({
  origin: function (origin, callback) {
    if (!origin) return callback(null, true);
    if (allowedOrigins.includes(origin)) {
      return callback(null, true);
    }
    return callback(new Error('Not allowed by CORS'));
  },
  credentials: true
}));

FAYDA (BENEFIT):
Production code se localhost default me puri tarah hata diya gaya hai. Koi bhi unauthorized
origin aakar session hijack ya cross-origin request nahi kar sakta.

2. FILE: backend/server.js (Global Error Handler & JSON Parsing Crash Fix)
--------------------------------------------------------------------------------
PROBLEM:
Pehle Express me koi central error handler nahi tha. Jab koi galat JSON bhejta tha
to server pura internal file path (/ExpertHub/backend/...) aur stack trace leak karta tha.

--- [PEHLE (BEFORE)] ---
// Koi central error handler nahi tha.
// SyntaxError hone par Express ka default HTML error page leak ho jata tha.
app.get('/api/health', (req, res) => {
  res.json({ status: 'ok', message: 'Backend is running!' });
});

--- [ABHI (NOW)] ---
app.get('/api/health', (req, res) => {
  res.json({ status: 'ok', message: 'Backend is running!' });
});

// Centralized Error Handling Middleware (Suppresses raw stack trace / path leakage)
app.use((err, req, res, next) => {
  if (res.headersSent) {
    return next(err);
  }

  // Handle JSON parsing / body-parser syntax errors cleanly
  if (err instanceof SyntaxError && err.status === 400 && 'body' in err) {
    return res.status(400).json({
      success: false,
      message: 'Invalid JSON request payload format.'
    });
  }

  // Handle CORS rejection
  if (err.message && err.message.includes('CORS')) {
    return res.status(403).json({
      success: false,
      message: 'Access denied: Origin not authorized by CORS policy.'
    });
  }

  // Safe Generic 500 error without leaking stack trace
  console.error('[Unhandled Error]:', err.message);
  res.status(err.status || 500).json({
    success: false,
    message: process.env.NODE_ENV === 'production' 
      ? 'An internal server error occurred.' 
      : (err.message || 'Server Error')
  });
});

FAYDA (BENEFIT):
Server paths, node_modules line numbers, aur sensitive database error details ab
kisi bhi attacker ko expose nahi honge. Clean JSON 400/500 response aayega.

const envFrontends = process.env.ALLOWED_ORIGINS ? process.env.ALLOWED_ORIGINS.split(',').map(s => s.trim()) : [];
const allowedFrontends = Array.from(new Set([...defaultFrontends, ...envFrontends])).filter(Boolean);
const incomingOrigin = req.get('origin') || req.get('referer');
let validatedBase = process.env.FRONTEND_URL || 'https://experthub.nextgoodtechnologies.in';

5. FILE: backend/controllers/contactController.js (Spam & Input Validation)
--------------------------------------------------------------------------------
PROBLEM:
Pehle contact form me bina email check kiye aur bina length validation ke email send
kar diya jata tha, jisse mail server spam abuse me fas sakta tha.

--- [PEHLE (BEFORE)] ---
const { name, email, subject, message } = req.body;
if (!name || !email || !message) {
  return res.status(400).json({ message: 'All fields are required' });
}

--- [ABHI (NOW)] ---
const { name, email, subject, message } = req.body;

if (!name || !email || !message) {
  return res.status(400).json({ success: false, message: 'Name, email and message are required.' });
}

// Strict email regex validation
const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
if (!emailRegex.test(email.trim())) {
  return res.status(400).json({ success: false, message: 'Please provide a valid email format.' });
}

// Length limit validation against DoS/buffer flooding
if (name.length > 100 || email.length > 150 || message.length > 3000) {
  return res.status(400).json({ success: false, message: 'Input text exceeds allowable character limit.' });
}

FAYDA (BENEFIT):
Spam bots invalid emails ya massive payload bhejkar server memory ya mail quota abuse
nahi kar sakte.

--------------------------------------------------------------------------------

Pehle public expert profile route me koi bhi invalid string bhejkar Mongoose CastError (500)
trigger karwaya ja sakta tha.

--- [PEHLE (BEFORE)] ---
router.get('/:id', async (req, res) => {
  try {
    const expert = await Expert.findById(req.params.id);
    if (expert) res.json(expert);
    else res.status(404).json({ message: 'Expert not found' });
  } catch (error) {
    res.status(500).json({ message: 'Server Error' });
  }
});

--- [ABHI (NOW)] ---
router.get('/:id', async (req, res) => {
  try {
    const { id } = req.params;
    if (!id || id === 'public' || !mongoose.Types.ObjectId.isValid(id)) {
      return res.status(404).json({ message: 'Expert not found' });
    }

    const expert = await Expert.findById(id)
      .select(PUBLIC_EXPERT_FIELDS)
      .populate('categoryId', 'name')
      .populate('subCategoryId', 'name');
    if (expert) res.json(expert);
    else res.status(404).json({ message: 'Expert not found' });
  } catch (error) {
    res.status(500).json({ message: 'Server Error' });
  }
});

FAYDA (BENEFIT):
Invalid IDs par 500 error nahi aayega, clean 404 response aayega.
